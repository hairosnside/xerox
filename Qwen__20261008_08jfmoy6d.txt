```html

        tailwind.config = {

    <style>

</head>
<body class="min-h-screen text-slate-200">

    <!-- Header -->
    <header class="glass sticky top-0 z-50 px-6 py-4 flex justify-between items-center border-b border-dark-border">
        <div class="flex items-center gap-3">
            <div class="w-10 h-10 bg-gradient-to-br from-sky-500 to-blue-600 rounded-lg flex items-center justify-center shadow-lg shadow-sky-500/20">
                <i class="fas fa-wave-square text-white text-xl"></i>
            </div>
            <div>
                <h1 class="text-xl font-bold tracking-wide text-white">P3: Paper Path Pulse</h1>
                <p class="text-xs text-slate-400">Real-Time Telemetry & Health Monitor</p>
            </div>
        </div>
        <div class="flex items-center gap-4">
            <div class="flex items-center gap-2 text-sm text-slate-400">
                <span class="w-2 h-2 bg-green-500 rounded-full pulse-dot"></span>
                <span id="status-text">Connecting...</span>
            </div>
            <button onclick="toggleSettings()" class="p-2 hover:bg-slate-700 rounded-lg transition-colors" title="Settings">
                <i class="fas fa-cog text-slate-300"></i>
            </button>
        </div>
    </header>

    <!-- Settings Modal -->
    <div id="settings-modal" class="fixed inset-0 bg-black/60 backdrop-blur-sm z-[60] hidden flex items-center justify-center">
        <div class="glass rounded-2xl p-6 w-full max-w-md shadow-2xl">
            <h2 class="text-lg font-bold mb-4 flex items-center gap-2"><i class="fas fa-sliders-h text-sky-400"></i> Log Configuration</h2>
            <label class="block text-sm text-slate-400 mb-2">Log File Path</label>
            <input type="text" id="log-path-input" class="w-full bg-slate-800 border border-slate-600 rounded-lg px-4 py-2 text-white focus:outline-none focus:ring-2 focus:ring-sky-500" placeholder="/var/log/paperpath.log">
            <div class="flex justify-end gap-3 mt-6">
                <button onclick="toggleSettings()" class="px-4 py-2 text-slate-400 hover:text-white">Cancel</button>
                <button onclick="saveSettings()" class="px-4 py-2 bg-sky-600 hover:bg-sky-500 text-white rounded-lg font-medium">Save & Apply</button>
            </div>
        </div>
    </div>

    <main class="p-6 max-w-7xl mx-auto space-y-6">
        
        <!-- Row 1: NTCLI Overview (30s Refresh) -->
        <section>
            <div class="flex items-center gap-2 mb-4">
                <i class="fas fa-server text-sky-400"></i>
                <h2 class="text-lg font-semibold">Printer Health Overview</h2>
                <span class="text-xs bg-slate-800 text-slate-400 px-2 py-0.5 rounded-full ml-2">Refresh: 30s</span>
            </div>
            <div class="grid grid-cols-1 md:grid-cols-3 gap-4">
                <div class="glass rounded-xl p-5">
                    <div class="flex justify-between items-start">
                        <div>
                            <p class="text-slate-400 text-sm">Status</p>
                            <p id="ntcli-status" class="text-2xl font-bold text-green-400 mt-1">Idle</p>
                        </div>
                        <i class="fas fa-heartbeat text-green-400 text-2xl"></i>
                    </div>
                </div>
                <div class="glass rounded-xl p-5">
                    <div class="flex justify-between items-start">
                        <div>
                            <p class="text-slate-400 text-sm">Supplies</p>
                            <p id="ntcli-supplies" class="text-2xl font-bold text-white mt-1">--</p>

                        <i class="fas fa-box-open text-sky-400 text-2xl"></i>
                    </div>

                <div class="glass rounded-xl p-5">
                    <div class="flex justify-between items-start">
                        <div>
                            <p class="text-slate-400 text-sm">Destinations</p>
                            <p id="ntcli-dest" class="text-2xl font-bold text-white mt-1">--</p>
                        </div>
                        <i class="fas fa-tray text-purple-400 text-2xl"></i>
                    </div>
                </div>
            </div>
        </section>

        <!-- Row 2: Image Position Error Chart (1s Refresh) -->
        <section>
            <div class="flex items-center gap-2 mb-4">
                <i class="fas fa-chart-line text-amber-400"></i>
                <h2 class="text-lg font-semibold">Image Position Error Analysis</h2>
                <span class="text-xs bg-slate-800 text-slate-400 px-2 py-0.5 rounded-full ml-2">Refresh: 1s</span>
                <div class="ml-auto flex gap-4 text-xs">
                    <span class="flex items-center gap-1"><span class="w-3 h-3 bg-blue-500 rounded-full"></span> Early (-)</span>

                </div>
                <div class="glass rounded-xl p-5 h-64 overflow-y-auto scrollbar-hide">
                    <ul id="qdmg-list" class="space-y-2 text-sm font-mono">
                        <li class="text-slate-500 italic">Waiting for QDMG data...</li>
                    </ul>
                </div>
            </div>

    <script>
        // --- Configuration ---
        let logPath = localStorage.getItem('logPath') || '/var/log/paperpath.log';
        document.getElementById('log-path-input').value = logPath;

        // --- Settings Modal ---
        function toggleSettings() {
            document.getElementById('settings-modal').classList.toggle('hidden');
        }
        function saveSettings() {
            logPath = document.getElementById('log-path-input').value.trim() || '/var/log/paperpath.log';
            localStorage.setItem('logPath', logPath);
            toggleSettings();
        }

        // --- Chart.js Initialization ---
        const ctx = document.getElementById('errorChart').getContext('2d');
        const sensors = ['Passtrhu', 'Printer Entry', 'S1 Sensor', 'Bump Nip', 'Bump Exit', '2ndXfer'];
        const colors = ['#38bdf8', '#818cf8', '#c084fc', '#f472b6', '#fb923c', '#facc15'];
        
        const datasets = sensors.map((sensor, i) => ({
            label: sensor,
            data: [],
            borderColor: colors[i],
            backgroundColor: colors[i] + '33',
            tension: 0.3,
            pointRadius: 0,
            borderWidth: 2
        }));

        const chart = new Chart(ctx, {
            type: 'line',
            data: { labels: [], datasets: datasets },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                animation: { duration: 0 },
                scales: {
                    x: { display: false },
                    y: { 
                        grid: { color: '#334155' }, 
                        ticks: { color: '#94a3b8' },
                        title: { display: true, text: 'Position Error (Units)', color: '#94a3b8' }
                    }
                },
                plugins: {
                    legend: { labels: { color: '#e2e8f0', boxWidth: 12, padding: 15 } },
                    tooltip: { mode: 'index', intersect: false }
                }
            }
        });

        // --- Data Fetching: NTCLI (30s) ---
        async function fetchNtcli() {
            try {
                const res = await fetch('/api/ntcli');
                const data = await res.json();
                if (data.ok) {
                    document.getElementById('ntcli-status').innerText = data.status?.State || 'Unknown';
                    document.getElementById('ntcli-supplies').innerText = data.supplies ? `${Object.keys(data.supplies).length} Items` : '0 Items';
                    document.getElementById('ntcli-dest').innerText = data.destination ? `${Object.keys(data.destination).length} Bins` : '0 Bins';
                    document.getElementById('status-text').innerText = 'Live';
                }
            } catch (e) {
                document.getElementById('status-text').innerText = 'Offline';
            }
        }

        // --- Data Fetching: Logs (1s) ---
        async function fetchLogs() {
            try {
                const res = await fetch(`/api/logs?path=${encodeURIComponent(logPath)}`);
                const json = await res.json();
                if (json.ok) {
                    updateChart(json.data.imageErrors);
                    updateList('slpq-list', json.data.slpq, 'SLPQ');
                    updateList('qdmg-list', json.data.qdmg, 'QDMG');
                }
            } catch (e) {
                console.error("Log fetch error", e);
            }
        }

        function updateChart(errors) {
            // Reset datasets
            datasets.forEach(ds => ds.data = []);
            chart.data.labels = [];
            
            // Map errors to datasets
            errors.forEach((err, index) => {
                chart.data.labels.push(index);
                const sensorIndex = sensors.indexOf(err.sensor);
                if (sensorIndex !== -1) {
                    datasets[sensorIndex].data.push(err.value);
                } else {
                    // Fill other datasets with null to maintain alignment
                    datasets.forEach((ds, i) => {
                        if (i !== sensorIndex) ds.data.push(null);
                    });
                }
            });
            chart.update();
        }

        function updateList(elementId, items, prefix) {
            const ul = document.getElementById(elementId);
            if (!items || items.length === 0) return;
            
            ul.innerHTML = items.slice(-15).reverse().map(item => {
                const time = new Date(item.time).toLocaleTimeString();
                return `<li class="p-2 bg-slate-800/50 rounded border-l-4 ${prefix === 'SLPQ' ? 'border-emerald-500' : 'border-rose-500'}">
                    <span class="text-slate-500 text-xs">[${time}]</span> ${item.raw}
                </li>`;
            }).join('');
        }

        // --- Intervals ---
        fetchNtcli();
        fetchLogs();
        setInterval(fetchNtcli, 30000); // 30 Seconds
        setInterval(fetchLogs, 1000);   // 1 Second
    </script>
</body>
